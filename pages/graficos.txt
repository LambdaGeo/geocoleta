import streamlit as st
import plotly.express as px
import pandas as pd
from core.registry import page

from core.filters import FilterEngine

#@page("Gráficos da Pesquisa")
def render_graficos(df: pd.DataFrame):

    st.title("📊 Gráficos - Dados")

    if df is None or df.empty:
        st.warning("Nenhum dado carregado.")
        return
    
    #filtros = FilterEngine(df)
    #df = filtros.apply()

    # Identifica colunas
    cols_numericas = df.select_dtypes(include="number").columns.tolist()
    cols_categoricas = df.select_dtypes(include="object").columns.tolist()

    st.markdown("### Configuração do Gráfico")
    col1, col2 = st.columns(2)

    with col1:
        tipo = st.selectbox("Tipo de gráfico", ["Barras", "Histograma"])

    with col2:
        cat = st.selectbox("Categoria", cols_categoricas)

    num = st.selectbox("Métrica", cols_numericas)

    # Renderização
    if tipo == "Barras":
        df_agg = df.groupby(cat)[num].mean().reset_index()
        fig = px.bar(df_agg, x=cat, y=num, color=cat)
    else:
        fig = px.histogram(df, x=num, color=cat)

    st.plotly_chart(fig, use_container_width=True)
